## 1. Alunos por Status de Matrícula

### 1.1 Filtrar alunos por status (pré-matriculado, matriculado, etc.)

```sql
-- Contar pré-matriculados vs outros status por turma
-- Fonte: bitts/Consultas-SQL (SQL025)
DECLARE
    @CODPERLET VARCHAR(10) = '2026',
    @CODFILIAL INT = 1;

SELECT
    A.CODTURMA,
    ISNULL([PRE_MATR].TOTAL, 0) AS TOTAL_PRE_MATRICULADOS,
    ISNULL([OUTROS].TOTAL, 0)   AS TOTAL_OUTROS
FROM SMATRICPL A (NOLOCK)
INNER JOIN SPLETIVO B (NOLOCK)
    ON B.IDPERLET = A.IDPERLET
    AND B.CODCOLIGADA = A.CODCOLIGADA
    AND B.CODFILIAL = A.CODFILIAL
LEFT JOIN (
    SELECT CODCOLIGADA, CODFILIAL, CODTURMA, IDHABILITACAOFILIAL, IDPERLET, COUNT(RA) AS TOTAL
    FROM SMATRICPL (NOLOCK)
    WHERE CODSTATUS IN (SELECT CODSTATUS FROM SSTATUS WHERE MATRICULAPROVISORIA = 'S')
    GROUP BY CODCOLIGADA, CODFILIAL, CODTURMA, IDHABILITACAOFILIAL, IDPERLET
) [PRE_MATR]
    ON A.CODCOLIGADA = [PRE_MATR].CODCOLIGADA
    AND A.CODFILIAL = [PRE_MATR].CODFILIAL
    AND A.CODTURMA = [PRE_MATR].CODTURMA
    AND A.IDPERLET = [PRE_MATR].IDPERLET
LEFT JOIN (
    SELECT CODCOLIGADA, CODFILIAL, CODTURMA, IDHABILITACAOFILIAL, IDPERLET, COUNT(RA) AS TOTAL
    FROM SMATRICPL (NOLOCK)
    WHERE CODSTATUS IN (SELECT CODSTATUS FROM SSTATUS WHERE PLATIVO = 'S')
    GROUP BY CODCOLIGADA, CODFILIAL, CODTURMA, IDHABILITACAOFILIAL, IDPERLET
) [OUTROS]
    ON A.CODCOLIGADA = [OUTROS].CODCOLIGADA
    AND A.CODFILIAL = [OUTROS].CODFILIAL
    AND A.CODTURMA = [OUTROS].CODTURMA
    AND A.IDPERLET = [OUTROS].IDPERLET
WHERE B.CODPERLET = @CODPERLET AND A.CODFILIAL = @CODFILIAL
GROUP BY A.CODTURMA, [PRE_MATR].TOTAL, [OUTROS].TOTAL;
```

### 1.2 Listar alunos com situação de matrícula detalhada

```sql
-- Fonte: bitts/Consultas-SQL (SQL026)
SELECT
    MPL.RA,
    P.NOME,
    P.CPF,
    SS.DESCRICAO AS SITUACAO_MATRICULA,
    SS.PLATIVO,
    SS.MATRICULAPROVISORIA,
    SS.PLREMATRICULA,
    SS.PLINDICATRANC,
    MPL.CODTURMA,
    PL.CODPERLET,
    PL.DESCRICAO AS PERIODO_LETIVO,
    TM.DESCRICAO AS TIPO_MATRICULA
FROM SMATRICPL MPL (NOLOCK)
INNER JOIN SALUNO A (NOLOCK)
    ON MPL.CODCOLIGADA = A.CODCOLIGADA AND MPL.RA = A.RA
INNER JOIN PPESSOA P (NOLOCK)
    ON A.CODPESSOA = P.CODIGO
INNER JOIN SSTATUS SS (NOLOCK)
    ON MPL.CODCOLIGADA = SS.CODCOLIGADA AND MPL.CODSTATUS = SS.CODSTATUS
INNER JOIN SPLETIVO PL (NOLOCK)
    ON MPL.CODCOLIGADA = PL.CODCOLIGADA AND MPL.IDPERLET = PL.IDPERLET
LEFT JOIN STIPOMATRICULA TM (NOLOCK)
    ON MPL.CODCOLIGADA = TM.CODCOLIGADA AND MPL.CODTIPOMAT = TM.CODTIPOMAT
WHERE MPL.CODCOLIGADA = :CODCOLIGADA
    AND PL.CODPERLET = :CODPERLET
ORDER BY MPL.CODTURMA, P.NOME;
```

## 2. Matriculados por Turma/Disciplina

### 2.1 Listar alunos matriculados em uma turma com dados completos

```sql
-- Fonte: bitts/Consultas-SQL (SQL019)
SELECT
    MPL.RA,
    P.NOME,
    CONVERT(VARCHAR(10), P.DTNASCIMENTO, 103) AS DT_NASCIMENTO,
    DATEDIFF(YEAR, P.DTNASCIMENTO, GETDATE()) AS IDADE,
    P.SEXO,
    T.CODTURMA,
    T.NOME AS TURMA,
    D.NOME AS DISCIPLINA,
    PROF.NOME AS PROFESSOR,
    SS.DESCRICAO AS SITUACAO,
    MPL.NUMALUNO
FROM SMATRICPL MPL (NOLOCK)
INNER JOIN SALUNO A (NOLOCK)
    ON MPL.CODCOLIGADA = A.CODCOLIGADA AND MPL.RA = A.RA
INNER JOIN PPESSOA P (NOLOCK)
    ON A.CODPESSOA = P.CODIGO
INNER JOIN STURMA T (NOLOCK)
    ON MPL.CODCOLIGADA = T.CODCOLIGADA
    AND MPL.CODTURMA = T.CODTURMA
    AND MPL.IDPERLET = T.IDPERLET
    AND MPL.CODFILIAL = T.CODFILIAL
INNER JOIN SSTATUS SS (NOLOCK)
    ON MPL.CODCOLIGADA = SS.CODCOLIGADA AND MPL.CODSTATUS = SS.CODSTATUS
LEFT JOIN STURMADISC TD (NOLOCK)
    ON T.CODCOLIGADA = TD.CODCOLIGADA
    AND T.CODTURMA = TD.CODTURMA
    AND T.IDPERLET = TD.IDPERLET
LEFT JOIN SDISCIPLINA D (NOLOCK)
    ON TD.CODCOLIGADA = D.CODCOLIGADA AND TD.CODDISC = D.CODDISC
LEFT JOIN SPROFESSORTURMA PT (NOLOCK)
    ON TD.CODCOLIGADA = PT.CODCOLIGADA AND TD.IDTURMADISC = PT.IDTURMADISC
LEFT JOIN SPROFESSOR PROF (NOLOCK) ON PT.CODCOLIGADA = PROF.CODCOLIGADA AND PT.CODPROF = PROF.CODPROF
INNER JOIN PPESSOA PROFP (NOLOCK) ON PROF.CODPESSOA = PROFP.CODIGO
WHERE MPL.CODCOLIGADA = :CODCOLIGADA
    AND SS.PLATIVO = 'S'
ORDER BY MPL.NUMALUNO;
```

### 2.2 Alunos matriculados por disciplina (com JOIN completo)

```sql
-- Fonte: Fórum RM (topic/12424)
SELECT
    M.RA,
    P.NOME,
    D.NOME AS DISCIPLINA,
    C.NOME AS CURSO,
    PL.CODPERLET,
    T.CODTURMA,
    SS.DESCRICAO AS SITUACAO
FROM SMATRICULA M (NOLOCK)
INNER JOIN STURMADISC TD (NOLOCK)
    ON M.CODCOLIGADA = TD.CODCOLIGADA AND M.IDTURMADISC = TD.IDTURMADISC
INNER JOIN SDISCIPLINA D (NOLOCK)
    ON TD.CODCOLIGADA = D.CODCOLIGADA AND TD.CODDISC = D.CODDISC
INNER JOIN STURMA T (NOLOCK)
    ON TD.CODCOLIGADA = T.CODCOLIGADA AND TD.CODTURMA = T.CODTURMA
    AND TD.IDPERLET = T.IDPERLET AND TD.CODFILIAL = T.CODFILIAL
INNER JOIN SPLETIVO PL (NOLOCK)
    ON TD.CODCOLIGADA = PL.CODCOLIGADA AND TD.IDPERLET = PL.IDPERLET
INNER JOIN SHABILITACAOFILIAL HF (NOLOCK)
    ON TD.CODCOLIGADA = HF.CODCOLIGADA AND TD.IDHABILITACAOFILIAL = HF.IDHABILITACAOFILIAL
INNER JOIN SCURSO C (NOLOCK)
    ON HF.CODCOLIGADA = C.CODCOLIGADA AND HF.CODCURSO = C.CODCURSO
INNER JOIN SALUNO A (NOLOCK)
    ON M.CODCOLIGADA = A.CODCOLIGADA AND M.RA = A.RA
INNER JOIN PPESSOA P (NOLOCK)
    ON A.CODPESSOA = P.CODIGO
INNER JOIN SSTATUS SS (NOLOCK)
    ON M.CODCOLIGADA = SS.CODCOLIGADA AND M.CODSTATUS = SS.CODSTATUS
WHERE M.CODCOLIGADA = :CODCOLIGADA
    AND PL.CODPERLET = :CODPERLET
ORDER BY D.NOME, P.NOME;
```

## 3. Notas e Boletim

### 3.1 Mapa de Notas por Turma

```sql
-- Baseado em bitts/Consultas-SQL (SQL016)
SELECT
    A.RA,
    P.NOME,
    D.NOME AS DISCIPLINA,
    NE.CODETAPA,
    E.DESCRICAO AS ETAPA,
    NE.NOTAFALTA AS NOTA
FROM SNOTAETAPA NE (NOLOCK)
INNER JOIN SETAPAS E (NOLOCK)
    ON NE.CODCOLIGADA = E.CODCOLIGADA
    AND NE.IDTURMADISC = E.IDTURMADISC
    AND NE.CODETAPA = E.CODETAPA
    AND NE.TIPOETAPA = E.TIPOETAPA
INNER JOIN STURMADISC TD (NOLOCK)
    ON NE.CODCOLIGADA = TD.CODCOLIGADA AND NE.IDTURMADISC = TD.IDTURMADISC
INNER JOIN SDISCIPLINA D (NOLOCK)
    ON TD.CODCOLIGADA = D.CODCOLIGADA AND TD.CODDISC = D.CODDISC
INNER JOIN SALUNO A (NOLOCK)
    ON NE.CODCOLIGADA = A.CODCOLIGADA AND NE.RA = A.RA
INNER JOIN PPESSOA P (NOLOCK)
    ON A.CODPESSOA = P.CODIGO
WHERE NE.CODCOLIGADA = :CODCOLIGADA
    AND NE.TIPOETAPA = 'N'  -- Apenas notas (não faltas)
    AND TD.CODTURMA = :CODTURMA
ORDER BY P.NOME, D.NOME, NE.CODETAPA;
```

### 3.2 Faltas por Aluno/Disciplina

```sql
SELECT
    A.RA,
    P.NOME,
    D.NOME AS DISCIPLINA,
    NE.CODETAPA,
    E.DESCRICAO AS ETAPA,
    NE.NOTAFALTA AS FALTAS,
    E.AULASDADAS,
    CASE WHEN E.AULASDADAS > 0
        THEN ROUND(((E.AULASDADAS - NE.NOTAFALTA) * 100.0 / E.AULASDADAS), 1)
        ELSE 0
    END AS PERCENTUAL_PRESENCA
FROM SNOTAETAPA NE (NOLOCK)
INNER JOIN SETAPAS E (NOLOCK)
    ON NE.CODCOLIGADA = E.CODCOLIGADA
    AND NE.IDTURMADISC = E.IDTURMADISC
    AND NE.CODETAPA = E.CODETAPA
    AND NE.TIPOETAPA = E.TIPOETAPA
INNER JOIN STURMADISC TD (NOLOCK)
    ON NE.CODCOLIGADA = TD.CODCOLIGADA AND NE.IDTURMADISC = TD.IDTURMADISC
INNER JOIN SDISCIPLINA D (NOLOCK)
    ON TD.CODCOLIGADA = D.CODCOLIGADA AND TD.CODDISC = D.CODDISC
INNER JOIN SALUNO A (NOLOCK)
    ON NE.CODCOLIGADA = A.CODCOLIGADA AND NE.RA = A.RA
INNER JOIN PPESSOA P (NOLOCK)
    ON A.CODPESSOA = P.CODIGO
WHERE NE.CODCOLIGADA = :CODCOLIGADA
    AND NE.TIPOETAPA = 'F'  -- Apenas faltas
    AND A.RA = :RA
ORDER BY D.NOME, NE.CODETAPA;
```

## 4. Turmas do Professor

```sql
SELECT
```

## 5. Financeiro — Parcelas do Aluno

```sql
SELECT
    A.RA,
    P.NOME,
    PA.PARCELA,
    PA.COTA,
    PA.DTVENCIMENTO,
    PA.VALOR,
    F.STATUSLAN,
    CASE F.STATUSLAN
        WHEN 0 THEN 'Aberto'
        WHEN 1 THEN 'Pago'
        WHEN 2 THEN 'Cancelado'
    END AS STATUS_DESCRICAO,
    F.VALORORIGINAL,
    F.VALORDESCONTO,
    F.VALORPAGO,
    F.DTPAGAMENTO,
    B.NOME AS BOLSA,
    BL.VALOR AS VALOR_BOLSA
FROM SPARCELA PA (NOLOCK)
INNER JOIN SALUNO A (NOLOCK)
    ON PA.CODCOLIGADA = A.CODCOLIGADA AND PA.RA = A.RA
INNER JOIN PPESSOA P (NOLOCK)
    ON A.CODPESSOA = P.CODIGO
LEFT JOIN SLAN SL (NOLOCK)
    ON PA.CODCOLIGADA = SL.CODCOLIGADA AND PA.IDPARCELA = SL.IDPARCELA
LEFT JOIN FLAN F (NOLOCK)
    ON SL.CODCOLIGADA = F.CODCOLIGADA AND SL.IDLAN = F.IDLAN
LEFT JOIN SBOLSALAN BL (NOLOCK)
    ON PA.CODCOLIGADA = BL.CODCOLIGADA AND PA.IDPARCELA = BL.IDPARCELA
LEFT JOIN SBOLSA B (NOLOCK)
    ON BL.CODCOLIGADA = B.CODCOLIGADA AND BL.CODBOLSA = B.CODBOLSA
WHERE PA.CODCOLIGADA = :CODCOLIGADA
    AND A.RA = :RA
ORDER BY PA.DTVENCIMENTO;
```

## 6. Inadimplência — Dashboard

```sql
-- Alunos inadimplentes (parcelas vencidas abertas)
SELECT
    A.RA,
    P.NOME,
    COUNT(F.IDLAN) AS PARCELAS_VENCIDAS,
    SUM(F.VALORORIGINAL - ISNULL(F.VALORDESCONTO, 0)) AS VALOR_TOTAL_DEVIDO,
    MIN(F.DTVENCIMENTO) AS VENCIMENTO_MAIS_ANTIGO,
    DATEDIFF(DAY, MIN(F.DTVENCIMENTO), GETDATE()) AS DIAS_ATRASO
FROM FLAN F (NOLOCK)
INNER JOIN SLAN SL (NOLOCK)
    ON F.CODCOLIGADA = SL.CODCOLIGADA AND F.IDLAN = SL.IDLAN
INNER JOIN SPARCELA PA (NOLOCK)
    ON SL.CODCOLIGADA = PA.CODCOLIGADA AND SL.IDPARCELA = PA.IDPARCELA
INNER JOIN SALUNO A (NOLOCK)
    ON PA.CODCOLIGADA = A.CODCOLIGADA AND PA.RA = A.RA
INNER JOIN PPESSOA P (NOLOCK)
    ON A.CODPESSOA = P.CODIGO
WHERE F.CODCOLIGADA = :CODCOLIGADA
    AND F.STATUSLAN = 0  -- Aberto
    AND F.DTVENCIMENTO < GETDATE()  -- Vencido
GROUP BY A.RA, P.NOME
ORDER BY DIAS_ATRASO DESC;
```

## 7. Histórico Escolar do Aluno

```sql
SELECT
    HA.RA,
    C.NOME AS CURSO,
    H.NOME AS HABILITACAO,
    PL.CODPERLET AS PERIODO,
    D.NOME AS DISCIPLINA,
    NE_NOTA.NOTAFALTA AS NOTA,
    NE_FALTA.NOTAFALTA AS FALTAS,
    SS.DESCRICAO AS RESULTADO,
    M.DTMATRICULA
FROM SMATRICULA M (NOLOCK)
INNER JOIN STURMADISC TD (NOLOCK)
    ON M.CODCOLIGADA = TD.CODCOLIGADA AND M.IDTURMADISC = TD.IDTURMADISC
INNER JOIN SDISCIPLINA D (NOLOCK)
    ON TD.CODCOLIGADA = D.CODCOLIGADA AND TD.CODDISC = D.CODDISC
INNER JOIN SPLETIVO PL (NOLOCK)
    ON TD.CODCOLIGADA = PL.CODCOLIGADA AND TD.IDPERLET = PL.IDPERLET
INNER JOIN SHABILITACAOFILIAL HF (NOLOCK)
    ON M.CODCOLIGADA = HF.CODCOLIGADA AND M.IDHABILITACAOFILIAL = HF.IDHABILITACAOFILIAL
INNER JOIN SCURSO C (NOLOCK)
    ON HF.CODCOLIGADA = C.CODCOLIGADA AND HF.CODCURSO = C.CODCURSO
INNER JOIN SHABILITACAO H (NOLOCK)
    ON HF.CODCOLIGADA = H.CODCOLIGADA AND HF.CODCURSO = H.CODCURSO AND HF.CODHABILITACAO = H.CODHABILITACAO
INNER JOIN SHABILITACAOALUNO HA (NOLOCK)
    ON M.CODCOLIGADA = HA.CODCOLIGADA AND M.RA = HA.RA AND M.IDHABILITACAOFILIAL = HA.IDHABILITACAOFILIAL
LEFT JOIN SSTATUS SS (NOLOCK)
    ON M.CODCOLIGADA = SS.CODCOLIGADA AND M.CODSTATUSRES = SS.CODSTATUS
LEFT JOIN (
    SELECT CODCOLIGADA, RA, IDTURMADISC, CODETAPA, NOTAFALTA
    FROM SNOTAETAPA (NOLOCK)
    WHERE TIPOETAPA = 'N' AND CODETAPA = (SELECT MAX(CODETAPA) FROM SETAPAS WHERE ETAPAFINAL = 'S')
) NE_NOTA
    ON M.CODCOLIGADA = NE_NOTA.CODCOLIGADA AND M.RA = NE_NOTA.RA AND M.IDTURMADISC = NE_NOTA.IDTURMADISC
LEFT JOIN (
    SELECT CODCOLIGADA, RA, IDTURMADISC, SUM(NOTAFALTA) AS NOTAFALTA
    FROM SNOTAETAPA (NOLOCK)
    WHERE TIPOETAPA = 'F'
    GROUP BY CODCOLIGADA, RA, IDTURMADISC
) NE_FALTA
    ON M.CODCOLIGADA = NE_FALTA.CODCOLIGADA AND M.RA = NE_FALTA.RA AND M.IDTURMADISC = NE_FALTA.IDTURMADISC
WHERE HA.CODCOLIGADA = :CODCOLIGADA
    AND HA.RA = :RA
ORDER BY PL.CODPERLET, D.NOME;
```

## 8. Validação e Introspecção

### 8.1 Verificar status configurados

```sql
SELECT
    CODSTATUS,
    DESCRICAO,
    PLATIVO,
    MATRICULAPROVISORIA,
    PLREMATRICULA,
    PLINDICATRANC,
    CUCONCCURSO,
    CUINDICATRANSF,
    CUINDICADESVINCULACAO,
    PLEXIGECONTRATO,
    PLCANCELACONTRATO,
    CUPERMITEMATRICPL
FROM SSTATUS (NOLOCK)
WHERE CODCOLIGADA = :CODCOLIGADA
ORDER BY CODSTATUS;
```

### 8.4 Contar alunos por status (dashboard)

```sql
SELECT
    SS.CODSTATUS,
    SS.DESCRICAO,
    COUNT(MPL.RA) AS TOTAL_ALUNOS
FROM SMATRICPL MPL (NOLOCK)
INNER JOIN SSTATUS SS (NOLOCK)
    ON MPL.CODCOLIGADA = SS.CODCOLIGADA AND MPL.CODSTATUS = SS.CODSTATUS
INNER JOIN SPLETIVO PL (NOLOCK)
    ON MPL.CODCOLIGADA = PL.CODCOLIGADA AND MPL.IDPERLET = PL.IDPERLET
WHERE MPL.CODCOLIGADA = :CODCOLIGADA
    AND PL.CODPERLET = :CODPERLET
GROUP BY SS.CODSTATUS, SS.DESCRICAO
ORDER BY TOTAL_ALUNOS DESC;
```
